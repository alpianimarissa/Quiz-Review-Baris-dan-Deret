        let currentState = {
            currentIndex: 0,
            userAnswers: {}, // index -> optionIndex
            score: 0
        };

        const STORAGE_KEY = 'arithmetic_quiz_state_v2';

        function loadState() {
            const saved = localStorage.getItem(STORAGE_KEY);
            if (saved) {
                try {
                    currentState = JSON.parse(saved);
                } catch(e) {
                    console.error("Gagal memuat state dari local storage", e);
                }
            }
        }

        function saveState() {
            localStorage.setItem(STORAGE_KEY, JSON.stringify(currentState));
        }

        function initQuiz() {
            loadState();
            renderGridNav();
            loadQuestion(currentState.currentIndex);
            updateScoreDisplay();
        }

        function loadQuestion(index) {
            currentState.currentIndex = index;
            saveState();

            const q = questions[index];
            document.getElementById('question-indicator').innerText = `Soal ${index + 1} dari ${questions.length}`;
            document.getElementById('category-indicator').innerText = q.category;
            document.getElementById('progress-bar').style.width = `${((index + 1) / questions.length) * 100}%`;
            document.getElementById('question-text').innerHTML = q.question;

            const optionsContainer = document.getElementById('options-container');
            optionsContainer.innerHTML = '';

            const answered = currentState.userAnswers[index] !== undefined;
            const selectedOption = currentState.userAnswers[index];

            q.options.forEach((opt, optIdx) => {
                const btn = document.createElement('button');
                btn.className = `option-card w-full text-left p-4 rounded-xl border border-slate-200 font-medium flex items-center justify-between text-sm sm:text-base `;
                
                if (answered) {
                    if (opt.isCorrect) {
                        btn.className += ' bg-emerald-50 border-emerald-500 text-emerald-900 font-semibold';
                    } else if (selectedOption === optIdx && !opt.isCorrect) {
                        btn.className += ' bg-rose-50 border-rose-500 text-rose-900 font-semibold';
                    } else {
                        btn.className += ' bg-white text-slate-500 opacity-60';
                    }
                    btn.disabled = true;
                } else {
                    btn.className += ' bg-white hover:bg-slate-50 text-slate-700 cursor-pointer shadow-sm';
                    btn.onclick = () => selectOption(index, optIdx);
                }

                btn.innerHTML = `<span class="flex items-center gap-3"><strong class="text-indigo-600">${String.fromCharCode(65 + optIdx)}.</strong> ${opt.text}</span>`;
                optionsContainer.appendChild(btn);
            });

            const expContainer = document.getElementById('explanation-container');
            if (answered) {
                expContainer.classList.remove('hidden');
                const isCorrectUser = q.options[selectedOption].isCorrect;
                expContainer.className = `mt-6 p-4 rounded-xl border text-sm ${isCorrectUser ? 'bg-emerald-50 border-emerald-200 text-emerald-900' : 'bg-rose-50 border-rose-200 text-rose-900'}`;
                document.getElementById('explanation-title').innerText = isCorrectUser ? 'Jawaban Anda Benar!' : 'Jawaban Kurang Tepat!';
                document.getElementById('explanation-text').innerHTML = q.options[selectedOption].rationale;
            } else {
                expContainer.classList.add('hidden');
            }

            document.getElementById('prev-btn').disabled = index === 0;
            document.getElementById('next-btn').innerText = index === questions.length - 1 ? 'Selesai' : 'Selanjutnya';

            renderGridNav();

            if (window.renderMathInElement) {
                renderMathInElement(document.getElementById('question-container'));
                if (answered) renderMathInElement(expContainer);
            }
        }

        function selectOption(qIdx, optIdx) {
            if (currentState.userAnswers[qIdx] !== undefined) return;

            currentState.userAnswers[qIdx] = optIdx;
            if (questions[qIdx].options[optIdx].isCorrect) {
                currentState.score++;
            }
            saveState();
            updateScoreDisplay();
            loadQuestion(qIdx);
        }

        function nextQuestion() {
            if (currentState.currentIndex < questions.length - 1) {
                loadQuestion(currentState.currentIndex + 1);
            } else {
                showFinalScore();
            }
        }

        function prevQuestion() {
            if (currentState.currentIndex > 0) {
                loadQuestion(currentState.currentIndex - 1);
            }
        }

        function jumpTo(index) {
            loadQuestion(index);
        }

        function updateScoreDisplay() {
            document.getElementById('score-display').innerText = `Skor: ${currentState.score} / ${questions.length}`;
